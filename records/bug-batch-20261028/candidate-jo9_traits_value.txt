if $args[0] = 'dealer_quote':
 local $jo9_dealer_vendor = $args[1]

 local jo9_dealer_brand = dyneval('result = ' + $jo9_dealer_actor + "['brand']")
 ! JO9 MOD END RUMOR_GUILD_QUOTE_ACTOR_20261028
 result = 0
 if $jo9_dealer_actor = 'slave': killvar 'is_stock'
 if $jo9_dealer_vendor ! 'farid' and $jo9_dealer_vendor ! 'wong': exit
 if jo9_dealer_state ! CONST_INT['slave_exist'] and jo9_dealer_state ! CONST_INT['slave_faint']: exit
 if snk_room_target > 0: exit
 if difficulty_setting = 2 and $jo9_dealer_vendor = 'farid' and jo9_dealer_brand = 0: is_stock = 1
 ! The rumor probe receives explicit appraisal inputs from the isolated scan.
 if $jo9_dealer_actor = 'slave':
  gs '$slave_price', $jo9_dealer_actor
  jo9_dealer_rating = rating
  jo9_dealer_base = slave_base_price
  jo9_dealer_price = slaveprice
 end
 ! Read derived values after the existing appraisal recalculates them.
 local jo9_dealer_obedience = dyneval('result = ' + $jo9_dealer_actor + "['obedience']")
 local $jo9_dealer_psy = $dyneval('$result = $' + $jo9_dealer_actor + '_psy_status')
 if difficulty_setting = 0 or difficulty_setting = 1:
  ! Native attribute price already includes versatile; apply it once to the independent grade candidate too.
  local jo9_dealer_grade_price = func('jo9_traits_value', 'base_price', $jo9_dealer_actor, common_price[jo9_dealer_rating])
  local jo9_dealer_value = max(30, jo9_dealer_base * 3 / 2, jo9_dealer_grade_price)
  if jo9_dealer_obedience < 1: jo9_dealer_value = jo9_dealer_value * 20 / 100
  ! Keep the original Wong broken-state depreciation, separate from current obedience.
  if $jo9_dealer_vendor = 'wong' and $jo9_dealer_psy = 'broken': jo9_dealer_value -= 4 * jo9_dealer_value / 5
  result = jo9_dealer_value * iif(difficulty_setting = 1, 90, 75) / 100
 else
  if $jo9_dealer_vendor = 'farid': result = max(10, jo9_dealer_price / 2)
  if $jo9_dealer_vendor = 'wong':
   local jo9_dealer_value = auction_price[jo9_dealer_rating]
   if $jo9_dealer_psy = 'broken': jo9_dealer_value -= 4 * auction_price[jo9_dealer_rating] / 5
   result = func('jo9_traits_value', 'base_price', $jo9_dealer_actor, jo9_dealer_value)
  end
 end
 ! JO9 MOD START SALE_CAP_B_DEALER_QUOTE_20261026
 result = min(6000, result)
 ! JO9 MOD END SALE_CAP_B_DEALER_QUOTE_20261026
 exit
